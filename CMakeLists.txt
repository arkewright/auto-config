cmake_minimum_required(VERSION 3.30)
project(workload_kernel LANGUAGES C CXX)

set(CMAKE_CXX_STANDARD 26)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(KCFG_LLVM_VERSION "23.1.4" CACHE STRING "Exact Clang and LLVM SDK version")
option(KCFG_BUILD_PLUGIN "Build the Clang front-end plugin" ON)
option(KCFG_BUILD_PROCESSOR "Build the host processor and terminal review" ON)
include(CTest)
if(KCFG_BUILD_PLUGIN)
  add_subdirectory(plugin)
endif()
if(KCFG_BUILD_PROCESSOR)
  add_subdirectory(processor)
endif()
