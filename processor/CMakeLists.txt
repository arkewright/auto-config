cmake_minimum_required(VERSION 3.25)
project(kcfg_processor LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 26)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(PROJECT_IS_TOP_LEVEL)
  include(CTest)
endif()

set(FTXUI_SOURCE_DIR "" CACHE PATH "Local FTXUI checkout (otherwise use installed package)")
if(NOT TARGET ftxui::component)
  if(NOT FTXUI_SOURCE_DIR AND EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/../.deps/ftxui/CMakeLists.txt")
    set(FTXUI_SOURCE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../.deps/ftxui")
  endif()
  if(FTXUI_SOURCE_DIR)
    set(CMAKE_POLICY_DEFAULT_CMP0077 NEW)
    set(FTXUI_ENABLE_INSTALL OFF)
    set(FTXUI_QUIET ON)
    add_subdirectory("${FTXUI_SOURCE_DIR}" "${CMAKE_CURRENT_BINARY_DIR}/ftxui" EXCLUDE_FROM_ALL)
  else()
    find_package(ftxui CONFIG REQUIRED)
  endif()
endif()

add_library(kcfg_plan_core OBJECT src/Inputs.cpp src/Decision.cpp)
target_include_directories(kcfg_plan_core PUBLIC include)
add_executable(kcfg-plan src/main.cpp src/Storage.cpp src/Review.cpp)
target_link_libraries(kcfg-plan PRIVATE kcfg_plan_core ftxui::component)
foreach(target IN ITEMS kcfg_plan_core kcfg-plan)
  if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Wconversion
      -Wsign-compare -Wold-style-cast -Wnon-virtual-dtor -Werror=return-type)
  endif()
endforeach()

if(BUILD_TESTING)
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  add_test(NAME kcfg-plan-contract COMMAND "${Python3_EXECUTABLE}" -I
    "${CMAKE_CURRENT_SOURCE_DIR}/tests/contract.py" "$<TARGET_FILE:kcfg-plan>")
endif()
